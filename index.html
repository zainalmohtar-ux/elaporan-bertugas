// Jalankan fungsi ciptaSheetDalamFolder() sekali di editor Apps Script untuk terus menjana Google Sheet di dalam folder Drive anda!
function ciptaSheetDalamFolder() {
  var FOLDER_ID = '15KBs80ThK0oeHX1C6jS_1pQVb-TBKICy';
  var SHEET_FILENAME = 'Pangkalan_Data_eLaporan_SK_Seri_Chantek';
  var folder = DriveApp.getFolderById(FOLDER_ID);
  var ss;

  var existingFiles = folder.getFilesByName(SHEET_FILENAME);
  if (existingFiles.hasNext()) {
    ss = SpreadsheetApp.open(existingFiles.next());
  } else {
    ss = SpreadsheetApp.create(SHEET_FILENAME);
    DriveApp.getFileById(ss.getId()).moveTo(folder);
  }

  var sheet = ss.getSheetByName('Rekod_Mingguan') || ss.getSheets()[0];
  sheet.setName('Rekod_Mingguan');
  if (sheet.getLastRow() === 0) {
    sheet.appendRow([
      'Tarikh Hantar', 'Sekolah', 'Minggu', 'Bil Laporan Harian',
      'Purata Keceriaan (%)', 'Johan Kelas', 'Bil Isu Keselamatan',
      'Purata Kehadiran RMT (%)', 'Bil Sesi RMT', 'Pautan Fail Arkib Drive', 'Data JSON'
    ]);
  }
  Logger.log('BERJAYA! Pautan Google Sheet dalam folder anda: ' + ss.getUrl());
  return ss;
}

function doPost(e) {
  var ss = ciptaSheetDalamFolder();
  var folder = DriveApp.getFolderById('15KBs80ThK0oeHX1C6jS_1pQVb-TBKICy');
  var sheet = ss.getSheetByName('Rekod_Mingguan');

  var data = JSON.parse(e.postData.contents);
  var timestamp = Utilities.formatDate(new Date(), 'Asia/Kuala_Lumpur', 'yyyyMMdd_HHmmss');

  // Arkibkan fail JSON laporan mingguan ke Folder Google Drive SK Seri Chantek
  var jsonFileName = 'eLaporan_SK_Seri_Chantek_Minggu_' + data.week + '_' + timestamp + '.json';
  var archiveFile = folder.createFile(jsonFileName, JSON.stringify(data, null, 2), MimeType.PLAIN_TEXT);

  // Simpan gambar bukti pemantauan terus ke dalam Folder Google Drive sekolah
  if (data.photos && data.photos.length > 0) {
    data.photos.forEach(function(item, index) {
      if (item.dataUrl && item.dataUrl.indexOf('base64,') > -1) {
        var base64 = item.dataUrl.split('base64,')[1];
        var blob = Utilities.newBlob(
          Utilities.base64Decode(base64),
          'image/jpeg',
          'Bukti_' + item.category + '_M' + data.week + '_' + (index + 1) + '_' + timestamp + '.jpg'
        );
        folder.createFile(blob);
      }
    });
  }

  sheet.appendRow([
    new Date(),
    data.schoolName || 'SEKOLAH KEBANGSAAN SERI CHANTEK',
    'Minggu ' + data.week,
    data.dailyCount,
    data.avgKeceriaan + '%',
    data.bestClass,
    data.safetyIssuesCount,
    data.avgRMT + '%',
    data.rmtCount,
    archiveFile.getUrl(),
    JSON.stringify(data.payload)
  ]);

  return ContentService.createTextOutput(JSON.stringify({
    status: 'success',
    sheetUrl: ss.getUrl(),
    fileUrl: archiveFile.getUrl()
  })).setMimeType(ContentService.MimeType.JSON);
}

Pasang Aplikasi SK Seri Chantek

Pasang sistem e-Laporan Guru Bertugas SK Seri Chantek terus ke skrin utama telefon pintar anda supaya boleh dibuka seperti aplikasi mudah alih rasmi serta menyokong simpanan luar talian (offline).

🍎 Pengguna iOS (iPhone / iPad - Safari)
  1. Buka fail atau pautan aplikasi ini menggunakan pelayar Safari.
  2. Tekan ikon Kongsi / Share ⬆️ di bahagian bawah (atau atas) skrin.
  3. Skrol ke bawah dan pilih "Add to Home Screen" (Tambah ke Skrin Utama).
  4. Tekan Add (Tambah) di penjuru kanan atas. Ikon lencana S.K. Seri Chantek akan muncul di skrin utama anda.
🤖 Pengguna Android (Google Chrome)
  1. Buka aplikasi ini dalam pelayar Google Chrome.
  2. Tekan menu 3 Titik (⋮) di penjuru kanan atas pelayar.
  3. Pilih "Install app" (Pasang aplikasi) atau "Add to Home screen".
  4. Sahkan pemasangan — aplikasi akan dipasang terus ke senarai aplikasi telefon Android anda.
